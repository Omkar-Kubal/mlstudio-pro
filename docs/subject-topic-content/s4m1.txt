# Example: Linear Regression with one feature (diabetes dataset)
import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import load_diabetes
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression

# Load diabetes data (10 features); use only the 'bmi' feature for simplicity
X_full, y = load_diabetes(return_X_y=True)
X = X_full[:, np.newaxis, 2]  # feature index 2 = BMI
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, random_state=0)

model = LinearRegression()
model.fit(X_train, y_train)

print("Coefficient (slope):", model.coef_[0])
print("Intercept:", model.intercept_)

y_pred = model.predict(X_test)
plt.scatter(X_test, y_test, color='skyblue', label='Data')
plt.plot(X_test, y_pred, color='orange', linewidth=2, label='Fit')
plt.xlabel("BMI (normalized)")
plt.ylabel("Disease Progression")
plt.title("Linear Regression Fit")
plt.legend(); plt.show()
This code fits a linear model to predict the target from one feature and plots the data versus the fitted line. In a web frontend, you could use a similar approach: feed data to a chart and update the line when the model changes.

# Example: Polynomial Regression (degree=2) on synthetic data
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression

# Generate synthetic data (quadratic trend with noise)
np.random.seed(0)
X = np.linspace(-3, 3, 50)
y = 0.5*X**2 + X + 2 + np.random.randn(50)*3

# Prepare polynomial features
poly = PolynomialFeatures(degree=2, include_bias=False)
X_poly = poly.fit_transform(X.reshape(-1,1))

model = LinearRegression()
model.fit(X_poly, y)
y_pred = model.predict(X_poly)

plt.scatter(X, y, color='lightgreen', label='Data')
plt.plot(X, y_pred, color='red', linewidth=2, label='Degree 2 Fit')
plt.xlabel("X"); plt.ylabel("y"); plt.title("Polynomial Regression"); plt.legend()
plt.show()

15. Support Vector Regression (SVR)
Theory

SVR fits a function inside an ε-insensitive tube:

Errors inside ε are ignored

Only points outside matter (support vectors)

Key ideas:

Margin

Kernel trick

Convex optimization

Visualization / Animation

17. Gaussian Process Regression (GPR)
Theory

GPR is probabilistic regression:

Learns distribution over functions

Outputs mean + uncertainty

Kernel defines smoothness

Used when:

Data is small

Uncertainty matters

Visualization / Animation

Mean curve

Confidence band (±2σ)

Band widens in sparse regions

Code
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF

kernel = RBF(length_scale=1.0)
gpr = GaussianProcessRegressor(kernel=kernel)
gpr.fit(X, y)

y_mean, y_std = gpr.predict(X, return_std=True)
